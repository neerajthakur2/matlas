import express from "express";
const app = express();
const PORT = process.env.PORT || 5000;
app.use((req, res, next) => {
  console.log(`[${new Date().toLocaleDateString()}]${req.method}${req.url}`);
  next();
});
const customQuery = (req, res, next) => {
  console.log("Query:", req.query.age);
  next();
};
app.get("/", (req, res) => {
  res.send("<h1>Hello world</h1>");
});
app.get("/about", (req, res) => {
  res.send("<h1>About Page</h1>");
});
app.get("/login", customQuery, (req, res) => {
  let qer= req.query.age
  if( req.query.age<18){
  res.send(`<h1>Your age is less than 18 and it is:${qer}</h1>`);
  }

});
//custom middleware

//middleware
//runs on specific path
app.use("/admin", (req, res, next) => {
  console.log(`[${new Date().toLocaleDateString()}]${req.method}${req.url}`);
  next();
});
app.listen(PORT, () => {
  console.log(`Server is runnindsadg on port ${PORT}`);
});
